cmake_minimum_required(VERSION 3.24)
project(ImageProcessing VERSION 1.0.0 LANGUAGES CXX)

option(IMAGEPROCESSING_ENABLE_CUDA "Build GPU implementations" ON)
option(IMAGEPROCESSING_LEGACY_PTX "Use forward-compatible PTX with CUDA 12.0-12.7" OFF)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(imageprocessing_cpu src/convolution.cpp src/image_io.cpp)
target_include_directories(imageprocessing_cpu PUBLIC include)
target_include_directories(imageprocessing_cpu SYSTEM PRIVATE third_party)
target_compile_options(imageprocessing_cpu PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)

if(IMAGEPROCESSING_ENABLE_CUDA)
    # Set the architecture before enabling CUDA so CMake's compiler checks use it too.
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        if(IMAGEPROCESSING_LEGACY_PTX)
            set(CMAKE_CUDA_ARCHITECTURES 86-virtual CACHE STRING "PTX JIT target")
        else()
            set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "RTX 5060 Ti target")
        endif()
    endif()
    if(NOT CMAKE_CUDA_COMPILER)
        if(DEFINED ENV{CUDACXX})
            set(CMAKE_CUDA_COMPILER "$ENV{CUDACXX}" CACHE FILEPATH "CUDA compiler")
        endif()
        find_program(IMAGEPROCESSING_NVCC nvcc)
        if(NOT CMAKE_CUDA_COMPILER AND IMAGEPROCESSING_NVCC)
            set(CMAKE_CUDA_COMPILER "${IMAGEPROCESSING_NVCC}" CACHE FILEPATH "CUDA compiler")
        endif()
    endif()
    if(NOT CMAKE_CUDA_COMPILER)
        message(FATAL_ERROR "CUDA compiler unavailable. Install CUDA Toolkit 12.8+ for RTX 5060 Ti, or use -DIMAGEPROCESSING_ENABLE_CUDA=OFF for CPU-only builds.")
    endif()
    # Check nvcc before architecture-dependent compiler tests for a useful mismatch error.
    execute_process(COMMAND "${CMAKE_CUDA_COMPILER}" --version OUTPUT_VARIABLE NVCC_VERSION_TEXT)
    if(NVCC_VERSION_TEXT MATCHES "release ([0-9]+\\.[0-9]+)")
        set(NVCC_RELEASE "${CMAKE_MATCH_1}")
        if(NVCC_RELEASE VERSION_LESS 12.0)
            message(FATAL_ERROR "CUDA Toolkit 12.0+ is required; native RTX 5060 Ti builds need 12.8+.")
        endif()
        if(NOT IMAGEPROCESSING_LEGACY_PTX AND NVCC_RELEASE VERSION_LESS 12.8)
            message(FATAL_ERROR "Found CUDA ${NVCC_RELEASE}. Download CUDA Toolkit 12.8+ from https://developer.nvidia.com/cuda-downloads for native sm_120 compilation. Optional compatibility build: -DIMAGEPROCESSING_LEGACY_PTX=ON (PTX JIT; requires a Blackwell-capable driver).")
        endif()
    endif()
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_library(imageprocessing_gpu src/gpu.cu)
    target_include_directories(imageprocessing_gpu PUBLIC include)
    target_link_libraries(imageprocessing_gpu PUBLIC imageprocessing_cpu CUDA::cudart)
    set_target_properties(imageprocessing_gpu PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    if(IMAGEPROCESSING_LEGACY_PTX)
        message(STATUS "PTX compatibility mode: driver JIT compilation; not native sm_120")
    endif()
endif()

add_executable(image_filters src/main.cpp)
target_link_libraries(image_filters PRIVATE imageprocessing_cpu)
if(IMAGEPROCESSING_ENABLE_CUDA)
    target_link_libraries(image_filters PRIVATE imageprocessing_gpu)
    target_compile_definitions(image_filters PRIVATE IMAGEPROCESSING_HAS_CUDA=1)
    if(IMAGEPROCESSING_LEGACY_PTX)
        target_compile_definitions(image_filters PRIVATE IMAGEPROCESSING_BUILD_MODE="ptx_compatibility")
    else()
        target_compile_definitions(image_filters PRIVATE IMAGEPROCESSING_BUILD_MODE="native")
    endif()
    target_compile_definitions(image_filters PRIVATE IMAGEPROCESSING_CUDA_COMPILER_VERSION="${CMAKE_CUDA_COMPILER_VERSION}")
else()
    target_compile_definitions(image_filters PRIVATE IMAGEPROCESSING_BUILD_MODE="cpu_only" IMAGEPROCESSING_CUDA_COMPILER_VERSION="none")
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(cpu_tests tests/cpu_tests.cpp)
    target_link_libraries(cpu_tests PRIVATE imageprocessing_cpu)
    add_test(NAME cpu_correctness COMMAND cpu_tests)
    if(IMAGEPROCESSING_ENABLE_CUDA)
        add_executable(gpu_tests tests/gpu_tests.cpp)
        target_link_libraries(gpu_tests PRIVATE imageprocessing_gpu)
        add_test(NAME gpu_correctness COMMAND gpu_tests)
        set_tests_properties(gpu_correctness PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endif()
